#+begin_src conf

profile {
    # left monitor: AOC 1 (secondary) [DP-1]

    output "Dell Inc. DELL SE2416H Y4GVX82704FL" mode 1920x1080@60Hz position 1920,0
    # right monitor: AOC 2 - Vertical (tertiary) [DVI-I-1]
    output "AOC 2369M 23039IA002615" mode 1920x1080@60Hz position 3840,0 transform 90
}

** MSc presentation
Masters presentation @ FEEC/Unicamp, with projector and Educart.
#+begin_src conf
# Room:
#
#   +----------------------+
#   |                      |
#   |    +-----+ +-----+   |
#   |    | tv1 | | tv2 |   |
#   |    +-----+ +-----+   |
#   |                      |
#   |     +--+     +--+    |
#   |   x |  |  3  |  | x  |
#   |   x |  |     |  | x  |
#   |   x |  |     |  | x  |
#   |   x |  |     |  | x  |
#   |      \  \___/  /     |
#   |    x  \___ 1 _/  x   |
#   |           x          |
#   |                      |
#   |       === 2 ===      |
#   +----------------------+
#
#   x: seats
#
#   1: laptop     native screen
#   2: projector  native HDMI --> EPSON projector
#   3: educart    USB-C HDMI --> Dell monitor
#
profile msc_presentation {
    # laptop
    output "eDP-1"    mode 1920x1080@60Hz position 0,0     scale 1
    # projector
    output "HDMI-A-1" mode 1920x1200@60Hz position 0,1080  scale 1
    # educart
    output "DP-1"     mode 1920x1080@60Hz position 0,-1080 scale 1
}
#+end_src
