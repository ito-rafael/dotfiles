** Template
#+name: install
#+begin_src emacs-lisp :var pin=0 comma=1 extension_id="aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa" :results output :tangle no
(princ (format "\"%s\": {\n" extension_id))
(when (= pin 1)
  (princ "\"toolbar_pin\": \"force_pinned\",\n"))
(princ "\"installation_mode\": \"normal_installed\",\n")
(princ "\"update_url\": \"https://clients2.google.com/service/update2/crx\"")
(if (= comma 1)
    (princ "},\n")
  (princ "}\n"))
#+end_src
